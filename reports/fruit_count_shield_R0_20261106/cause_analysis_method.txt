def rot(axis,a):
 axis=np.asarray(axis,float);axis/=np.linalg.norm(axis);x,y,z=axis;K=np.array([[0,-z,y],[z,0,-x],[-y,x,0]])
 return np.eye(3)+np.sin(a)*K+(1-np.cos(a))*(K@K)
def fk(q,link='eg2_base'):
 if link=='base_link':return np.eye(4)
 j=joints[link];o=j.find('origin');v=np.array([float(s) for s in o.get('xyz','0 0 0').split()]);r,p,y=[float(s) for s in o.get('rpy','0 0 0').split()]
 A=np.eye(4);A[:3,:3]=rot([0,0,1],y)@rot([0,1,0],p)@rot([1,0,0],r);A[:3,3]=v
 if j.get('type')=='revolute':
  B=np.eye(4);B[:3,:3]=rot([float(s) for s in j.find('axis').get('xyz').split()],q[int(j.get('name').split('_')[1])-1]);A=A@B
 return fk(q,j.find('parent').get('link'))@A
f=np.array([0,.045,.225]);f/=np.linalg.norm(f);right=np.cross(f,[0,-1,0]);right/=np.linalg.norm(right);up=np.cross(right,f)

def cam(q):
 T=fk(q);R=T[:3,:3]@CV;pos=T[:3,3]+T[:3,:3]@EYE;return pos,R

bc=np.array([x[0] for x in basket]);bh=np.array([x[1] for x in basket])
def raycast(origin,directions,cs=centers,hs=halves):
 directions=np.asarray(directions)
 with np.errstate(divide='ignore',invalid='ignore'):
  a=(cs[None]-hs[None]-origin[None,None])/directions[:,None,:]
  b=(cs[None]+hs[None]-origin[None,None])/directions[:,None,:]
 enter=np.max(np.minimum(a,b),axis=2);leave=np.min(np.maximum(a,b),axis=2)
 values=np.where((leave>=enter)&(leave>0),np.where(enter>0,enter,leave),np.inf)
 ids=np.argmin(values,axis=1);depth=values[np.arange(len(ids)),ids]
 return depth,ids
